\section*{Sigmoid Function}
\[\sigma(x) = \frac{1}{1 + e^{-x}} \]
\section*{ReLU Function}
\[f(x) = \max(0, x) \]
\section*{Tanh Function}
\[\tanh(x) = \frac{e^x - e^{-x}}{e^x + e^{-x}} \]
\section*{Softmax Function}
\[\text{softmax}(z_i) = \frac{e^{z_i}}{\sum_{j} e^{z_j}} \]
